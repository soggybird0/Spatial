Melee Slash (Hybrid)

Server applies damage; client plays immediate feedback.

Server

local Server = require(ReplicatedStorage.Spatial).Server

local function slash(player: Player)
    local character = player.Character
    if not character then return end
    local root = character:FindFirstChild("HumanoidRootPart") :: BasePart?
    if not root then return end

    local hitbox = Server:Create({
        Shape = "Box",
        Size = Vector3.new(5, 5, 7),
        Origin = root,
        Offset = CFrame.new(0, 0, -3.5),
        Duration = 0.2,
        Mode = "Once",
        Authority = "Hybrid",
        Owner = player,
        Filter = {
            LookingFor = "Humanoid",
            Blacklist = { character },
        },
        DotProduct = 0.2,
    })

    hitbox.Hit:Connect(function(result)
        if result.Humanoid then
            result.Humanoid:TakeDamage(30)
        end
    end)

    hitbox:Start()
end

Client

local Client = require(ReplicatedStorage.Spatial).Client

local function slash(character: Model)
    local root = character:FindFirstChild("HumanoidRootPart") :: BasePart?
    if not root then return end

    local hitbox = Client:Create({
        Shape = "Box",
        Size = Vector3.new(5, 5, 7),
        Origin = root,
        Offset = CFrame.new(0, 0, -3.5),
        Duration = 0.2,
        Authority = "Client",
    })

    hitbox.Hit:Connect(function(result)
        playSlashVFX(result.Position, result.Part)
    end)

    hitbox:Start()
end

Sphere AOE (Server Authority)

local Server = require(ReplicatedStorage.Spatial).Server

local function explosion(position: Vector3, radius: number, owner: Player?)
    local hitbox = Server:Create({
        Shape = "Sphere",
        Radius = radius,
        CFrame = CFrame.new(position),
        Duration = 0.05,
        Mode = "Pierce",
        Authority = "Server",
        Owner = owner,
        Filter = { LookingFor = "Humanoid" },
    })

    hitbox.Hit:Connect(function(result)
        if result.Humanoid then
            result.Humanoid:TakeDamage(50)
        end
    end)

    hitbox:Start()
end

Welded Weapon

local Server = require(ReplicatedStorage.Spatial).Server

local function attachToWeapon(player: Player, weaponPart: BasePart)
    local hitbox = Server:Create({
        Shape = "Box",
        Size = Vector3.new(1, 1, 4),
        Duration = 0,
        Mode = "Cooldown",
        Debounce = 0.15,
        Authority = "Hybrid",
        Owner = player,
        Filter = {
            LookingFor = "Humanoid",
            Blacklist = { player.Character },
        },
    })

    hitbox:WeldTo(weaponPart, CFrame.new(0, 0, -2))
    hitbox.Hit:Connect(function(result)
        if result.Humanoid then
            result.Humanoid:TakeDamage(15)
        end
    end)

    hitbox:Start()
    -- later: hitbox:Stop() / hitbox:Destroy()
end

Projectile (Server)

local Server = require(ReplicatedStorage.Spatial).Server

local function fireProjectile(player: Player, origin: CFrame, speed: number)
    local part = Instance.new("Part")
    part.Size = Vector3.new(0.5, 0.5, 2)
    part.CFrame = origin
    part.Anchored = true
    part.CanCollide = false
    part.Parent = workspace

    local hitbox = Server:Create({
        Shape = "Part",
        Part = part,
        Duration = 2,
        Mode = "Once",
        Authority = "Server",
        Owner = player,
        Filter = {
            LookingFor = "Humanoid",
            Blacklist = { player.Character },
        },
    })

    hitbox.Hit:Connect(function(result)
        if result.Humanoid then
            result.Humanoid:TakeDamage(40)
        end
        hitbox:Destroy()
        part:Destroy()
    end)

    hitbox:Start()

    local conn
    conn = game:GetService("RunService").Heartbeat:Connect(function(dt)
        if not part.Parent then conn:Disconnect() return end
        part.CFrame += part.CFrame.LookVector * speed * dt
    end)
end

Tracking NPCs

local Server = require(ReplicatedStorage.Spatial).Server

Server:Track(npcModel)

npcModel.Destroying:Connect(function()
    Server:Untrack(npcModel)
end)
Without tracking Lag compensation cannot rewind that model. Always track potential targets.
