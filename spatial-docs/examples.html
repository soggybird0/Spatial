Melee slash (Hybrid)

Server damages; client plays VFX.

Server

local Server = require(ReplicatedStorage.Spatial).Server

local function slash(player: Player)
    local root = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
    if not root then return end

    local hitbox = Server:Create({
        Shape = "Box", Size = Vector3.new(5, 5, 7),
        Origin = root, Offset = CFrame.new(0, 0, -3.5),
        Duration = 0.2, Mode = "Once", Authority = "Hybrid",
        Owner = player,
        Filter = { LookingFor = "Humanoid", Blacklist = { player.Character } },
        DotProduct = 0.2,
    })
    hitbox.Hit:Connect(function(r)
        if r.Humanoid then r.Humanoid:TakeDamage(30) end
    end)
    hitbox:Start()
end

Client

local Client = require(ReplicatedStorage.Spatial).Client
-- same geometry; Hit → VFX only
hitbox.Hit:Connect(function(r) playSlashVFX(r.Position) end)

Welded weapon

local hitbox = Server:Create({
    Shape = "Box", Size = Vector3.new(1, 1, 4),
    Duration = 0, Mode = "Cooldown", Debounce = 0.15,
    Authority = "Hybrid", Owner = player,
    Filter = { LookingFor = "Humanoid", Blacklist = { player.Character } },
})
hitbox:WeldTo(weaponPart, CFrame.new(0, 0, -2))
hitbox:Start()

Projectile

local part = Instance.new("Part")
part.Size = Vector3.new(0.5, 0.5, 2)
part.CFrame = origin; part.Anchored = true; part.CanCollide = false
part.Parent = workspace

local hitbox = Server:Create({
    Shape = "Part", Part = part, Duration = 2,
    Mode = "Once", Authority = "Server", Owner = player,
    Filter = { LookingFor = "Humanoid", Blacklist = { player.Character } },
})
hitbox.Hit:Connect(function(r)
    if r.Humanoid then r.Humanoid:TakeDamage(40) end
    hitbox:Destroy(); part:Destroy()
end)
hitbox:Start()
